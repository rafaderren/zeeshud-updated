// Controls the team panels animating in match start

event HudTournament_AnimateTeams_In
{
	Animate RedTeamPanel xpos c150 Spline 2.5 0.5
	Animate BlueTeamPanel xpos c-300 Spline 2.5 0.5
}

// This also hides the black background behind team panels

event HudTournament_AnimateTeams_Out
{
	Animate RedTeamPanel xpos r-5 Accel 0 0.68
	Animate BlueTeamPanel xpos -155 Accel 0 0.68
	Animate PlayersFooter ypos 480 Spline 0 0.5
	Animate PlayersHeader ypos -50 Spline 0 0.5
	SetVisible	PlayersBG	0	0
}

{	
	SetVisible	PlayersHeader	1	0
	SetVisible	PlayersFooter	1	0
	SetVisible	DoorLeft	1	0
	SetVisible	DoorLeftHinge	1	0
	SetVisible	DoorLeftCenter	1	0
	SetVisible	DoorLeftMiddle	1	0
	SetVisible	DoorLeftDecorBottom	1	0
	SetVisible	DoorLeftDecorTop	1	0
	SetVisible	DoorRight	1	0
	SetVisible	DoorRightCenter	1	0
	SetVisible	DoorRightHinge	1	0
	SetVisible	DoorRightMiddle	1	0
	SetVisible	DoorRightDecorBottom	1	0
	SetVisible	DoorRightDecorTop	1	0
	SetVisible	BlueTeamPanel	1	0
	SetVisible	RedTeamPanel	1	0
	SetVisible	PlayersBG	1	2
	SetVisible CountdownLabel 1 0
	SetVisible CountdownLabelShadow 1 0
	Animate CountdownLabel ypos 150 Linear 0 0
	Animate CountdownLabelShadow ypos 150 Linear 0 0
	Animate DoorLeftAnchor xpos 325 Spline 0 0.45
	Animate DoorRightAnchor xpos 1228 Spline 0 0.45 

	PlaySound 0 "player/taunt_yeti_standee_swoosh.wav"
	PlaySound 0.3 "ui/mm_door_close.wav"
	PlaySound 2 "ui/mm_door_open.wav"
	PlaySound 2.8 "ui/trade_changed.wav"
	PlaySound 7.5 "ui/trade_changed.wav"

	Animate CountdownLabel ypos c-3 Gain 0.75 0 0.4
	Animate CountdownLabelShadow ypos c-0 Gain 0.75 0 0.4

	RunEventChild FrontParticlePanel PlayDoorSlamParticles 0.45

	SetVisible	HudTournamentBG 0	2
	Animate DoorLeftAnchor xpos -165 Accel 2 1.5
	Animate DoorRightAnchor xpos 1750 Accel 2 1.5
	Animate PlayersFooter ypos 431 Spline 2.5 0.5
	Animate PlayersHeader ypos 0 Spline 2.5 0.5

	RunEvent HudTournament_AnimateTeams_In 0.25
	RunEvent HudTournament_AnimateTeams_Out 7.70

	SetVisible CountdownLabel 0 9.9
	SetVisible CountdownLabelShadow 0 9.9

	RunEvent HudTournament_DoorsCleanup 10
	
	SetVisible CountdownBG 1 11
}
